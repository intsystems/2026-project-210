    \small
    \textbf{Mixture of experts:}
    $
    \mathbf{z} = \sum_{e=1}^{E} g_e(\mathbf{x})\,\mathrm{FFN}_e(\mathbf{x}),~~~
    \mathrm{FFN}_e(\mathbf{x})=\mathbf{W}_e^{(2)}\mathrm{ReLU}(\mathbf{W}_e^{(1)}\mathbf{x}+\mathbf{b}_e^{(1)})+\mathbf{b}_e^{(2)}
    $
    Гейтинг: $g_e(\mathbf{x}) = \dfrac{\exp(\mathbf{a}_e^\top\mathbf{x}+b_e)}{\sum_j\exp(\mathbf{a}_j^\top\mathbf{x}+b_j)}$ (softmax) или top-$K$.
    Параметры $\mathbf{w}=(\mathbf{w}_{\text{exp}},\mathbf{w}_g)$.

    \vspace{0.1cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Разложение Гессе:}
            \\
            $\quad\quad\quad\quad\quad\mathbf{H}=\mathbf{G}+\mathbf{R}$, $\mathbf{G}=\mathbf{J}^{\top}\mathbf{A}\mathbf{J}$
            
            \textbf{Блочная факторизация:}
            \\
            $\quad\quad\quad\quad\mathbf{G}^{(p,q)}=\mathbf{P}^{(p)\top}\mathbf{S}^{(p)\top}\mathbf{A}\mathbf{S}^{(q)}\mathbf{P}^{(q)}$

            \textbf{Цель:} верхняя оценка $\|\mathbf{G}\|_2$ для MoE.
